-- ===== FIX the Week 2 share measures: the selector now filters through segment_dim =====
Share of customers = DIVIDE ( [Customers], CALCULATE ( [Customers], REMOVEFILTERS ( segments ), REMOVEFILTERS ( segment_dim ) ) )
Share of deal value = DIVIDE ( [Deal value], CALCULATE ( [Deal value], REMOVEFILTERS ( segments ), REMOVEFILTERS ( segment_dim ) ) )

-- ===== DAY 11 · Segment cards with a comparison sub-line =====
Conversion % (all customers) = CALCULATE ( [Conversion % (by close date)], REMOVEFILTERS ( segments ), REMOVEFILTERS ( segment_dim ) )
Late payments % (all customers) = CALCULATE ( [Late payments %], REMOVEFILTERS ( segments ), REMOVEFILTERS ( segment_dim ) )
Median days to decide = MEDIAN ( clean_leads[days_to_decision] )
Median days to decide (all customers) = CALCULATE ( [Median days to decide], REMOVEFILTERS ( segments ), REMOVEFILTERS ( segment_dim ) )
Price level =
    VAR buyers = COUNT ( clean_customers[price_level_gbp] )
    VAR renters = COUNT ( clean_customers[rent_level_pcm] )
    RETURN IF ( buyers >= renters,
        "£" & FORMAT ( MEDIAN ( clean_customers[price_level_gbp] ) / 1000, "#,0" ) & "k",
        "£" & FORMAT ( MEDIAN ( clean_customers[rent_level_pcm] ), "#,0" ) & " pcm" )
-- Sub-lines (put each in a small text card under its KPI card)
Conversion sub-line = "vs " & FORMAT ( [Conversion % (all customers)], "0.0%" ) & " for all customers"
Late sub-line = "vs " & FORMAT ( [Late payments % (all customers)], "0.0%" ) & " for all customers"
Days sub-line = "vs " & FORMAT ( [Median days to decide (all customers)], "0" ) & " days for all customers"
Value sub-line = FORMAT ( [Share of customers], "0.0%" ) & " of customers"
-- Channel share for "How they arrive" (use with segment_channels, filter channel_rank <= 5)
Channel share of segment leads = DIVIDE ( SUM ( segment_channels[leads] ), CALCULATE ( SUM ( segment_channels[leads] ), REMOVEFILTERS ( segment_channels[channel], segment_channels[channel_group], segment_channels[channel_rank] ) ) )

-- ===== DAY 12 · Tier tag, action panel =====
Tier = SELECTEDVALUE ( segment_tiers[priority_tier] )
Tier colour = SWITCH ( SELECTEDVALUE ( segment_tiers[priority_tier] ), "P1", "#E8702A", "P2", "#1F4E79", "P3", "#9BA7B4" )
Recommended action = SELECTEDVALUE ( segment_actions[recommended_action], "Select one segment to see its recommended action" )
Action: size and value = SELECTEDVALUE ( segment_actions[size_and_value] )
Action: conversion = SELECTEDVALUE ( segment_actions[conversion_fact] )
Action: behaviour = SELECTEDVALUE ( segment_actions[behaviour_fact] )
Open leads (table) = SUM ( open_leads_by_tier[open_leads] )

-- ===== DAY 13 · Segment movement =====
Customers moved % =
    DIVIDE ( CALCULATE ( SUM ( segment_movement[customers] ), segment_movement[stayed] = FALSE () ),
             SUM ( segment_movement[customers] ) )

-- ===== DAY 14 · Test KPI card (KPI visual: Value, Trend axis = test_kpi_weekly[week_starting], Target) =====
Test viewing 14d % =
    DIVIDE ( CALCULATE ( SUM ( test_kpi_weekly[viewing_within_14_days] ), test_kpi_weekly[test_group] = "Test" ),
             CALCULATE ( SUM ( test_kpi_weekly[leads_with_14_days] ), test_kpi_weekly[test_group] = "Test" ) )
Control viewing 14d % =
    DIVIDE ( CALCULATE ( SUM ( test_kpi_weekly[viewing_within_14_days] ), test_kpi_weekly[test_group] = "Control" ),
             CALCULATE ( SUM ( test_kpi_weekly[leads_with_14_days] ), test_kpi_weekly[test_group] = "Control" ) )
Test target % = [Control viewing 14d %] + MAX ( test_settings[target_uplift_pts] ) / 100
Test uplift (pts) = ( [Test viewing 14d %] - [Control viewing 14d %] ) * 100
